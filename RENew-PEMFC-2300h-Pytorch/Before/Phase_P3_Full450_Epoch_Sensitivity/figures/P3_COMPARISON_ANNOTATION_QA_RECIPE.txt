# Post-completion figure annotation QA only. No model, data, metric or frozen execution code change.
import sys
sys.path.insert(0,'scripts')
import p3_common as c
c.install_firewall()
import matplotlib
matplotlib.use('Agg')
import matplotlib.pyplot as plt
plt.rcParams.update({'font.family':['Arial','DejaVu Sans'],'font.size':13,'axes.spines.right':False,'axes.spines.top':False,'axes.linewidth':1.5,'legend.frameon':False})
data=c.pd.read_csv(c.OUT/'P3_P2_EARLY_STOP_VS_FULL450_COMPARISON.csv',float_precision='round_trip')
colors={'G0':'#767676','G1':'#B64342','G2':'#0F4D92'}
fig,axes=plt.subplots(1,3,figsize=(17,5))
for ax,seed in zip(axes,(42,52,62)):
    for g in ('G0','G1','G2'):
        row=data[data.model.eq(g)&data.seed.eq(seed)].iloc[0]
        ax.plot([0,1],[row.P2_RMSE_mV,row.P3_best_RMSE_mV],color=colors[g],marker='o',label=g,lw=2)
        offset=(-18 if g=='G2' else 16 if g=='G0' else 5) if seed==52 else (-16 if seed==62 and g=='G0' else 5)
        ax.annotate(f'P2 stop {int(row.P2_actual_stop_epoch)}',(0,row.P2_RMSE_mV),xytext=(4,offset),textcoords='offset points',fontsize=9,color=colors[g])
    ax.set_xticks([0,1],['P2 actual early stop','P3 best of 450'])
    ax.set_title(f'Seed {seed}')
    ax.set_ylim(bottom=0)
axes[0].set_ylabel('Formal query RMSE (mV)')
axes[-1].legend()
fig.suptitle('Separate runs, not a spliced training trajectory')
fig.tight_layout(pad=2)
for ext in ('png','pdf'):
    fig.savefig(c.PHASE/'figures'/('P3_P2_ACTUAL_STOP_COMPARISON.'+ext),dpi=300,facecolor='white')
plt.close(fig)
