 q=P/name;q.parent.mkdir(parents=True,exist_ok=True);q.write_bytes((json.dumps(obj,ensure_ascii=False,indent=2,allow_nan=False)+'\n').encode('utf-8'))

def same_numeric(a,b):
 a=np.asarray(a,dtype=np.float64);b=np.asarray(b,dtype=np.float64)
 return (a.view(np.uint64)==b.view(np.uint64))|(np.isnan(a)&np.isnan(b))
def independent_selector(meta):
 if len(meta)!=3007:raise ValueError('Historical logical prefix requires exactly3007 metadata rows')
 return np.flatnonzero(meta.bin_end_ophrs2_h.to_numpy()<CUTOFF)
def guarded_feature_extent(ids,meta):
 # This is a separate no-Test gate. Do not silently skip a Test row inside the mask.
 if len(ids)==0 or meta.iloc[ids]['split'].astype(str).str.contains('test',case=False).any():raise PermissionError('Test row would enter numeric parser')
 if not np.array_equal(ids,np.arange(len(ids))):raise PermissionError('Noncontiguous mask: requires an explicitly reviewed selective reader, not a prefix assumption')
 return len(ids)
def legal_origins(frame,cfg,start,end):
 # Independent window-identity implementation; no tensor/window values constructed.
 valid=frame.bin_valid_for_windows.to_numpy().astype(bool);segment=frame.temporal_segment_id.to_numpy()
 age=frame.bin_end_ophrs2_h.to_numpy();features=frame[cfg['task']['raw_f2_feature_order']].to_numpy()
 output=[]
 for o in range(start,end-24):
  f=o-47;l=o+24
  if f<0 or l>=end:continue
  sl=slice(f,l+1)
  if valid[sl].all() and (segment[sl]==segment[o]).all() and np.allclose(np.diff(age[sl]),.5,atol=1e-6,rtol=0) and np.isfinite(features[sl]).all():output.append(o)
 return output
